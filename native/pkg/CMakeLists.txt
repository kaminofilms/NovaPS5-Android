cmake_minimum_required(VERSION 3.22)
project(nova_pkg LANGUAGES CXX)
find_package(ZLIB REQUIRED)
add_library(nova_pkg STATIC package/pkgParser.cpp package/pfsParser.cpp package/Decompressor.cpp install.cpp)
target_compile_features(nova_pkg PUBLIC cxx_std_20)
target_include_directories(nova_pkg PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(nova_pkg PRIVATE KYTY_HAS_ZLIB=1)
target_link_libraries(nova_pkg PUBLIC ZLIB::ZLIB ${CMAKE_DL_LIBS})
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
  add_executable(nova_pkg_tests ../../tests/test_kytyplus_package.cpp)
  target_link_libraries(nova_pkg_tests PRIVATE nova_pkg)
  add_test(NAME nova_pkg_parser COMMAND nova_pkg_tests)
endif()

if(ANDROID)
  set_target_properties(nova_pkg PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_library(novapkg SHARED jni.cpp)
  set_target_properties(novapkg PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  target_link_libraries(novapkg PRIVATE nova_pkg)
  target_link_options(novapkg PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()
