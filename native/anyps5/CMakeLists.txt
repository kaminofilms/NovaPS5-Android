cmake_minimum_required(VERSION 3.28)
project(NovaAnyPS5 LANGUAGES C CXX)
set(ANYPS5_SOURCE_DIR "" CACHE PATH "Pinned AnyPS5 source checkout")
if(NOT EXISTS "${ANYPS5_SOURCE_DIR}/core/shader/recompiler/Recompiler.hpp")
    message(FATAL_ERROR "ANYPS5_SOURCE_DIR must name the pinned AnyPS5 checkout")
endif()
option(ANYPS5_ENABLE_SPIRV_TOOLS "Build complete upstream optimizer" ON)
add_library(nova_anyps5_host_headers INTERFACE)
function(nova_add_anyps5_compiler)
    # Upstream recompiler tests need AnyPS5's own test harness; Nova runs its own tests below.
    set(BUILD_TESTING OFF)
    set(PROJECT_SOURCE_DIR "${ANYPS5_SOURCE_DIR}")
    set(ANYPS5_SPIRV_HEADERS_SOURCE_DIR "${ANYPS5_SOURCE_DIR}/3rdparty/SPIRV-Headers")
    set(ANYPS5_SPIRV_HEADERS_INCLUDE_DIR "${ANYPS5_SPIRV_HEADERS_SOURCE_DIR}/include")
    set(ANYPS5_GLSLANG_SOURCE_DIR "${ANYPS5_SOURCE_DIR}/3rdparty/glslang")
    set(ANYPS5_GLSLANG_INCLUDE_DIRS "${ANYPS5_GLSLANG_SOURCE_DIR}")
    include("${ANYPS5_SOURCE_DIR}/core/shader/recompiler/CMakeLists.txt")
    add_shader_recompiler(nova_anyps5_compiler nova_anyps5_host_headers)
endfunction()
nova_add_anyps5_compiler()
add_library(nova_anyps5_shader SHARED shader_bridge.cpp)
target_link_libraries(nova_anyps5_shader PRIVATE nova_anyps5_compiler)
target_compile_features(nova_anyps5_shader PUBLIC cxx_std_20)
if(NOT APPLE)
    target_link_options(nova_anyps5_shader PRIVATE "-Wl,--exclude-libs,ALL")
endif()
target_include_directories(nova_anyps5_shader PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "$<TARGET_PROPERTY:nova_anyps5_compiler,INTERFACE_INCLUDE_DIRECTORIES>")
set_target_properties(nova_anyps5_shader PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(ANDROID)
    target_link_options(nova_anyps5_shader PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(nova_anyps5_compiler_test compiler_test.cpp)
    target_link_libraries(nova_anyps5_compiler_test PRIVATE nova_anyps5_shader)
    add_test(NAME anyps5_real_compute COMMAND nova_anyps5_compiler_test)
endif()
if(BUILD_TESTING)
    foreach(case IN ITEMS Float16Conversion Spirv13Image PositionInvariance)
        add_executable(nova_anyps5_${case} "${ANYPS5_SOURCE_DIR}/core/libs/prx/libSceAgcDriver/tests/${case}.cpp")
        target_link_libraries(nova_anyps5_${case} PRIVATE nova_anyps5_compiler)
        add_test(NAME anyps5_${case} COMMAND nova_anyps5_${case})
    endforeach()
endif()
if(ANDROID)
    target_sources(nova_anyps5_shader PRIVATE android_window.cpp)
    target_link_libraries(nova_anyps5_shader PRIVATE android)
endif()
